     * throw new IllegalArgumentException("A URL deve usar http ou https");

     * throw new IllegalArgumentException("Informe uma URL válida", e);
     * }
     * }

     * public String getLegenda() {
     * return legenda;

     */

    public void setImagemUrl(String imagemUrl) {
        if (imagemUrl == null || imagemUrl.isBlank()) {
            throw new IllegalArgumentException("A imagem é obrigatória.");
        }

        String valorLimpo = imagemUrl.trim();

        // Tenta validar se é um link da web (http ou https)
        if (isUrlValida(valorLimpo)) {
            this.imagemUrl = valorLimpo;
            return;
        }

        // Se não for URL, tenta validar se é um caminho de arquivo físico válido no
        // computador
        if (isArquivoLocalValido(valorLimpo)) {
            this.imagemUrl = valorLimpo;
            return;
        }

        throw new IllegalArgumentException("Informe uma URL válida ou um caminho de arquivo de imagem existente.");
    }

    // Auxiliar para validar URLs da web
    private boolean isUrlValida(String texto) {
        try {
            URI uri = new URI(texto);
            String esquema = uri.getScheme();
            return ("http".equalsIgnoreCase(esquema) || "https".equalsIgnoreCase(esquema))
                    && uri.getHost() != null;
        } catch (URISyntaxException e) {
            return false;
        }
    }

    // Auxiliar para validar se o arquivo físico realmente existe na máquina
    private boolean isArquivoLocalValido(String caminho) {
        try {
            Path path = Paths.get(caminho);
            // Verifica se o arquivo existe no disco e não é um diretório
            return Files.exists(path) && !Files.isDirectory(path);
        } catch (Exception e) {
            return false;
        }
    }

    public String getLegenda() {
        return legenda;
    }
    public void setLegenda(String legenda) {
        if (legenda != null && !legenda.isBlank()) {
            this.legenda = legenda;
        }
        // ou lançar exceção se for obrigatório
    }

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;


public class ServicoDeUpload {

    // Pasta onde os arquivos serão salvos no computador/servidor
    private static final String DIRETORIO_UPLOAD = "database/upload/";

    public String salvarArquivoNoDisco(byte[] bytesDoArquivo, String nomeOriginalDoArquivo) {
        try {
            // Garante que a pasta de uploads existe, senão ele cria
            Path caminhoDiretorio = Paths.get(DIRETORIO_UPLOAD);
            if (!Files.exists(caminhoDiretorio)) {
                Files.createDirectories(caminhoDiretorio);
            }

            // Define o caminho final onde o arquivo será guardado
            Path caminhoCompleto = caminhoDiretorio.resolve(System.currentTimeMillis() + "_" + nomeOriginalDoArquivo);

            // Salva os bytes fisicamente no disco rigido
            Files.write(caminhoCompleto, bytesDoArquivo);

            // Retorna a string do caminho relativo para salvar no objeto Postagem
            return caminhoCompleto.toString();
            
        } catch (Exception e) {
            throw new RuntimeException("Falha ao salvar o arquivo físico no computador: " + e.getMessage());
        }
    }
}

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

import org.mindrot.jbcrypt.BCrypt;

import com.projeto.croche.database_manager.DatabaseManager;

public class AuthService {
    /* Autenticação de usuárias */
    public void autenticarUsuario(String email, String senha) {
        System.out.println("Autenticando usuária...");

        boolean usuarioEncontrado = false;

        String sqlBusca = "SELECT u.id, u.senha " +
                     "FROM usuarios u " +
                     "WHERE u.email = ?";

         try (Connection conn = DatabaseManager.conectar(); 
         PreparedStatement pstmt = conn.prepareStatement(sqlBusca)) {
        
        pstmt.setString(1, email);

        try (ResultSet rs = pstmt.executeQuery()) {
            if (rs.next()) {
                String senhaHash = rs.getString("senha");
                // Verifica se a senha fornecida corresponde ao hash armazenado no banco de dados
                if (BCrypt.checkpw(senha, senhaHash)) {
                    int usuarioId = rs.getInt("id");
                    usuarioEncontrado = true;

                    System.out.println("Usuária autenticada com sucesso!");
                    System.out.println("ID: " + usuarioId);
                }
            }
            if (!usuarioEncontrado) {
                System.out.println("E-mail ou senha inválidos.");
            }
        }
    } catch (SQLException e) {
        System.out.println("Erro ao buscar dados: " + e.getMessage());
        return; // Sai do método se der erro na busca
    }

    }
}

package com.projeto.croche.validador_email;

public interface ValidadorEmail {
    boolean validar(String email);
}

package com.projeto.croche.validador_email;

public class RegexEmailAdapter implements ValidadorEmail  {
    @Override 
    public boolean validar(String email) {
        String regex = "^[A-Za-z0-9+_.-]+@(?:[A-Za-z0-9-]+\\.)+[A-Za-z]{2,}$";
        if (email == null|| email.isBlank() || !email.contains("@")) return false;
        return email.matches(regex);
    }
}

package com.projeto.croche.validador_email;

public class EmailService {

    private ValidadorEmail validador;

    public EmailService() {
        this(new RegexEmailAdapter());
    }

    public EmailService(ValidadorEmail validador) {
        this.validador = validador;
    }

    public boolean validarEmail(String email) {
        return validador.validar(email);
    }
}

package com.projeto.croche.database_manager;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;
import java.sql.Statement;

public class DatabaseManager {
    // Caminho onde o arquivo do banco de dados será salvo
    private static final String URL_CONEXAO = "jdbc:sqlite:database/croche.db";

    public static Connection conectar() {
        Connection conexao = null;

        // cria a pasta 'database' caso não exista
        try {
            // Garante que a pasta 'database' existe antes de conectar
            java.io.File pasta = new java.io.File("database");
            if (!pasta.exists()) {
                pasta.mkdirs();
            }
        } catch (Exception e) {
            System.out.println("Erro ao criar pasta do banco de dados: " + e.getMessage());
        }

        try {
            conexao = DriverManager.getConnection(URL_CONEXAO);
            System.out.println("Conexão com o SQLite estabelecida com sucesso!");
        } catch (SQLException e) {
            System.out.println("Erro ao conectar com o banco de dados: " + e.getMessage());
        }
        return conexao;
    }

    // Método para criar as tabelas iniciais automaticamente ao iniciar o app
    public static void criarTabelas() {
        String sqlUsuarios = "CREATE TABLE IF NOT EXISTS usuarios (" +
                "id INTEGER PRIMARY KEY AUTOINCREMENT, " +
                "nome TEXT NOT NULL, " +
                "email TEXT UNIQUE NOT NULL, " +
                "senha TEXT NOT NULL, " +
                "bio TEXT" +
                ");";

        String sqlPostagens = "CREATE TABLE IF NOT EXISTS postagens (" +
                "id INTEGER PRIMARY KEY AUTOINCREMENT, " +
                "usuario_id INTEGER NOT NULL, " +
                "imagem_url TEXT NOT NULL, " +
                "legenda TEXT, " +
                "FOREIGN KEY (usuario_id) REFERENCES usuarios(id)" +
                ");";

        String sqlCurtidas = "CREATE TABLE IF NOT EXISTS curtidas (" +
        // "id INTEGER PRIMARY KEY AUTOINCREMENT, " +
                "usuario_id INTEGER NOT NULL, " +
                "postagem_id INTEGER NOT NULL, " +
                "PRIMARY KEY (usuario_id, postagem_id), " +
                "FOREIGN KEY (usuario_id) REFERENCES usuarios(id), " +
                "FOREIGN KEY (postagem_id) REFERENCES postagens(id)" +
                ");";

        try (Connection conn = conectar();
                Statement stmt = conn.createStatement()) {

            stmt.execute(sqlUsuarios);
            stmt.execute(sqlPostagens);
            stmt.execute(sqlCurtidas);
            System.out.println("Tabelas verificadas/criadas com sucesso.");

        } catch (SQLException e) {
            System.out.println("Erro ao criar tabelas: " + e.getMessage());
        }
    }
}

package com.projeto.croche.dao;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.SQLException;

import org.mindrot.jbcrypt.BCrypt;

import com.projeto.croche.database_manager.DatabaseManager;
import com.projeto.croche.model.Usuario;

//Instância única da classe UsuarioDAO, método getInstance() retorna a instância única da classe, garantindo que apenas uma instância seja criada durante a execução do programa. Isso é útil para gerenciar o acesso aos dados do usuário de forma centralizada e consistente.
public class UsuarioDAO {
    private static class Holder {
        private static final UsuarioDAO INSTANCE = new UsuarioDAO();
    }

    private UsuarioDAO() {
    }

    public static UsuarioDAO getInstance() {
        return Holder.INSTANCE;
    }

    public void salvarUsuario(Usuario usuario) {
        salvarUsuario(usuario.getNome(), usuario.getEmail(), usuario.getSenhaHash(), usuario.getBio());
    }

    // Métodos para acessar e manipular os dados do usuário no banco de dados
    public void salvarUsuario(String nome, String email, String senha, String bio) {
        // Lógica para salvar o usuário no banco de dados
        String senhaHash = BCrypt.hashpw(senha, BCrypt.gensalt(12));
        try (Connection connection = DatabaseManager.conectar();
                PreparedStatement statement = connection.prepareStatement(
                        "INSERT INTO usuarios (nome, email, senha, bio) VALUES (?, ?, ?, ?)")) {
            statement.setString(1, nome);
            statement.setString(2, email);
            statement.setString(3, senhaHash);
            statement.setString(4, bio);
            statement.executeUpdate();
            // } catch (SQLException e) {
            // e.printStackTrace();
            // }
        } catch (SQLException e) {
            throw new RuntimeException("Erro ao salvar usuário", e);
        }
        System.out.println("Salvando usuário: " + nome + ", Email: " + email);
    }
}
